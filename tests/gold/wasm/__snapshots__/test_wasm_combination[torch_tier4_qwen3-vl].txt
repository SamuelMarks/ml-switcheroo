--- Forward ---
import jax.numpy as jnp
import flax.nnx as nnx

class Qwen3VLVisionConfig:
    in_channels: int = 3
    hidden_size: int = 1280
    temporal_patch_size: int = 2
    patch_size: int = 14

class Qwen3VLPatchEmbed(nnx.Module):
    '''3D Convolutional patch embedding for vision input.'''
    def __init__(self, rngs: nnx.Rngs, config: Qwen3VLVisionConfig):
        self.config = config
        kernel = (config.temporal_patch_size, config.patch_size, config.patch_size)
        self.proj = nnx.Conv(
            in_channels=config.in_channels, out_channels=config.hidden_size, kernel_size=kernel, stride=kernel, padding=0, bias=True, rngs=rngs)

    def __call__(self, hidden_states: jax.Array) -> jax.Array:
        cfg = self.config
        seq_len = hidden_states.shape[0]

        hidden_states = hidden_states.reshape(
            seq_len, cfg.in_channels, cfg.temporal_patch_size, cfg.patch_size, cfg.patch_size
        )

        out = self.proj(hidden_states)

        return out.reshape(seq_len, cfg.hidden_size)

import torch
import torch.nn as nn

class Qwen3VLVisionConfig:
    in_channels: int = 3
    hidden_size: int = 1280
    temporal_patch_size: int = 2
    patch_size: int = 14

class Qwen3VLPatchEmbed(nn.Module):
    '''3D Convolutional patch embedding for vision input.'''
    def __init__(self, config: Qwen3VLVisionConfig):
        super().__init__()
        self.config = config
        kernel = (config.temporal_patch_size, config.patch_size, config.patch_size)
        self.proj = nn.Conv2d(
            in_channels=config.in_channels, out_channels=config.hidden_size, kernel_size=kernel, stride=kernel, padding=0, bias=True)

    def forward(self, hidden_states: torch.Tensor) -> torch.Tensor:
        cfg = self.config
        seq_len = hidden_states.shape[0]

        hidden_states = hidden_states.reshape(
            seq_len, cfg.in_channels, cfg.temporal_patch_size, cfg.patch_size, cfg.patch_size
        )

        out = self.proj(hidden_states)

        return out.reshape(seq_len, cfg.hidden_size)
